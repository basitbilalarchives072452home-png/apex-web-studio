import * as THREE from 'three';
import { Terminal, Cpu, Globe, ArrowUpRight, Mail, Phone, ShieldCheck, Zap } from 'lucide-react';

    const scene = new THREE.Scene();
    const camera = new THREE.PerspectiveCamera(75, window.innerWidth / window.innerHeight, 0.1, 1000);
    const renderer = new THREE.WebGLRenderer({ alpha: true, antialias: true });
    
    renderer.setSize(window.innerWidth, window.innerHeight);
    renderer.setPixelRatio(Math.min(window.devicePixelRatio, 2));
    mountRef.current.appendChild(renderer.domElement);

    // Create 3D Glowing Particle Grid / Sphere
    const geometry = new THREE.IcosahedronGeometry(3, 2);
    const material = new THREE.MeshBasicMaterial({
      color: 0x06b6d4,
      wireframe: true,
      transparent: true,
      opacity: 0.15
    });
    const sphere = new THREE.Mesh(geometry, material);
    scene.add(sphere);

    // Particle field
    const particlesGeometry = new THREE.BufferGeometry();
    const particlesCount = 700;
    const posArray = new Float32Array(particlesCount * 3);

    for(let i = 0; i < particlesCount * 3; i++) {
      posArray[i] = (Math.random() - 0.5) * 15;
    }

    particlesGeometry.setAttribute('position', new THREE.BufferAttribute(posArray, 3));
    const particlesMaterial = new THREE.PointsMaterial({
      size: 0.02,
      color: 0x38bdf8
    });
    const particlesMesh = new THREE.Points(particlesGeometry, particlesMaterial);
    scene.add(particlesMesh);

    camera.position.z = 5;

    let mouseX = 0;
    let mouseY = 0;

    const handleMouseMove = (event) => {
      mouseX = event.clientX / window.innerWidth - 0.5;
      mouseY = event.clientY / window.innerHeight - 0.5;
    };

    window.addEventListener('mousemove', handleMouseMove);

    const animate = () => {
      requestAnimationFrame(animate);
      sphere.rotation.x += 0.002;
      sphere.rotation.y += 0.003;
      particlesMesh.rotation.y -= 0.001;

      sphere.rotation.x += mouseY * 0.05;
      sphere.rotation.y += mouseX * 0.05;

      renderer.render(scene, camera);
    };

    animate();

    const handleResize = () => {
      camera.aspect = window.innerWidth / window.innerHeight;
      camera.updateProjectionMatrix();
      renderer.setSize(window.innerWidth, window.innerHeight);
    };

    window.addEventListener('resize', handleResize);

    return () => {
      window.removeEventListener('mousemove', handleMouseMove);
      window.removeEventListener('resize', handleResize);
      if (mountRef.current) {
        mountRef.current.removeChild(renderer.domElement);
      }
    };
  }, []);

      <div ref={mountRef} className="fixed inset-0 pointer-events-none z-0" />

            A
          </div>
          <span className="text-xl font-extrabold tracking-wider bg-gradient-to-r from-white via-slate-200 to-cyan-400 bg-clip-text text-transparent">
            APEX WEB STUDIO
          </span>
        </div>
        <div className="hidden md:flex items-center space-x-8 text-sm text-slate-300 font-medium">
          <a href="#services" className="hover:text-cyan-400 transition">Capabilities</a>
          <a href="#work" className="hover:text-cyan-400 transition">Live Demos</a>
          <a href="#contact" className="hover:text-cyan-400 transition">Contact</a>
        </div>
        <a 
          href="#contact" 
          className="relative group px-5 py-2.5 rounded-full bg-cyan-500 text-slate-950 font-bold text-sm tracking-wide transition-all hover:bg-cyan-400 hover:shadow-[0_0_25px_rgba(6,182,212,0.6)]"
        >
          Initialize Project
        </a>
      </nav>

      {/* Hero Section */}
      <section className="relative z-10 flex flex-col items-center justify-center min-h-[85vh] text-center px-6 max-w-5xl mx-auto">
        <div className="inline-flex items-center space-x-2 px-3.5 py-1.5 rounded-full border border-cyan-500/30 bg-cyan-950/30 text-cyan-400 text-xs font-semibold tracking-wider uppercase mb-6 backdrop-blur-sm">
          <Zap className="w-3.5 h-3.5" />
          <span>Next-Gen Frontend Engineering</span>
        </div>
        
        <h1 className="text-4xl md:text-7xl font-black tracking-tight leading-tight mb-6 text-white">
          Architecting High-Speed <br />
          <span className="bg-gradient-to-r from-cyan-400 via-blue-500 to-indigo-500 bg-clip-text text-transparent">
            Digital Experiences
          </span>
        </h1>

        <p className="text-base md:text-xl text-slate-400 max-w-2xl mb-10 leading-relaxed">
          Founded by Bilal Basit. We engineer high-performance web apps, React/Next.js interfaces, and immersive digital platforms for global agencies.
        </p>

        <div className="flex flex-col sm:flex-row items-center justify-center gap-4 w-full">
          <a 
            href="#work" 
            className="w-full sm:w-auto px-8 py-4 rounded-xl bg-gradient-to-r from-cyan-500 to-blue-600 text-slate-950 font-bold text-sm tracking-wide hover:opacity-90 transition shadow-[0_0_30px_rgba(6,182,212,0.4)] flex items-center justify-center space-x-2"
          >
            <span>Explore Live Demos</span>
            <ArrowUpRight className="w-4 h-4" />
          </a>
          <a 
            href="#contact" 
            className="w-full sm:w-auto px-8 py-4 rounded-xl border border-slate-700 bg-slate-900/60 text-white font-bold text-sm tracking-wide hover:bg-slate-800 transition backdrop-blur-sm"
          >
            Direct Inquiry
          </a>
        </div>
      </section>

      {/* Services Section */}
      <section id="services" className="relative z-10 py-24 px-6 max-w-6xl mx-auto">
        <div className="text-center mb-16">
          <h2 className="text-3xl md:text-4xl font-black text-white mb-4">Core Capabilities</h2>
          <p className="text-slate-400 text-sm md:text-base">Built for speed, scale, and uncompromising visual standard.</p>
        </div>

        <div className="grid grid-cols-1 md:grid-cols-3 gap-8">
          <div className="p-8 rounded-2xl border border-slate-800 bg-slate-900/40 backdrop-blur-md hover:border-cyan-500/50 transition group">

              <Cpu className="w-6 h-6" />

          </div>
        </div>
      </section>

      {/* Contact Section */}
      <section id="contact" className="relative z-10 py-24 px-6 max-w-4xl mx-auto border-t border-slate-800/60">
        <div className="p-8 md:p-12 rounded-3xl border border-slate-800 bg-gradient-to-b from-slate-900/80 to-slate-950/80 backdrop-blur-xl text-center">
          <h2 className="text-3xl md:text-4xl font-black text-white mb-4">Let's Build Together</h2>
          <p className="text-slate-400 text-sm md:text-base mb-8 max-w-xl mx-auto">
            Ready to scale your agency's frontend capacity? Reach out directly via email or phone.
          </p>

          <div className="flex flex-col sm:flex-row items-center justify-center gap-6 text-slate-300 font-medium">
            <a 
              href="mailto:basitbilalarchives072452home@gmail.com" 
              className="flex items-center space-x-3 px-6 py-3.5 rounded-xl border border-slate-700 bg-slate-900 hover:border-cyan-500 transition"
            >
              <Mail className="w-5 h-5 text-cyan-400" />
              <span className="text-sm">basitbilalarchives072452home@gmail.com</span>
            </a>

            <a 
              href="tel:+923326471777" 
              className="flex items-center space-x-3 px-6 py-3.5 rounded-xl border border-slate-700 bg-slate-900 hover:border-cyan-500 transition"
            >
              <Phone className="w-5 h-5 text-cyan-400" />
              <span className="text-sm">+92 3326471777</span>
            </a>
          </div>
        </div>
      </section>

      {/* Footer */}
      <footer className="relative z-10 py-8 px-6 text-center border-t border-slate-900 text-slate-500 text-xs">
        <p>© {new Date().getFullYear()} Apex Web Studio. Founder: Bilal Basit. All rights reserved.</p>
      </footer>
    </div>
  );
}
